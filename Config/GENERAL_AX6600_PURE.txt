# PURE 版增量配置：保持轻量基础，预置 Tailscale 与 OpenClash。
# 本文件仅在 WRT_PROFILE=PURE 时追加到通用配置。

# Tailscale（WireGuard 内核模块已在 GENERAL_AX6600.txt 启用）
CONFIG_PACKAGE_tailscale=y
CONFIG_PACKAGE_luci-app-tailscale=y

# OpenClash
CONFIG_PACKAGE_luci-app-openclash=y
CONFIG_PACKAGE_luci-compat=y
# OpenClash 的 nftset 功能需要 dnsmasq-full；显式关闭精简 dnsmasq，
# 防止 defconfig 因上游默认配置变化而静默回退。
CONFIG_PACKAGE_dnsmasq-full=y
CONFIG_PACKAGE_dnsmasq=n
CONFIG_PACKAGE_dnsmasq_full_nftset=y
CONFIG_PACKAGE_dnsmasq_full_ipset=n
